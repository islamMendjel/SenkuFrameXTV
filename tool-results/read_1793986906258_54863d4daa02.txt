    32→    32→// ─── StreamPlayer · multi-server player w/ Autopilot precision sync ──────────
    33→    33→//
    34→    34→// VidSrc-family providers post PLAYER_EVENT telemetry (currentTime / duration
    35→    35→// / play-state) up the relay chain to our window. usePlayerSync turns that
    36→    36→// into a live clock → perfectly-synced subtitles, an exact transport bar and
    37→    37→// the end-of-episode Autopilot card. Servers that don't report fall back to
    38→    38→// a one-tap manual sync clock.
    39→    39→
    40→    40→interface Props {
    41→    41→  type: MediaType;
    42→    42→  imdbId: string;
    43→    43→  title: string;
    44→    44→  season?: number;
    45→    45→  episode?: number;
    46→    46→  episodeTitle?: string;
    47→    47→  poster?: string;
    48→    48→  runtimeMin?: number; // for progress estimation (manual mode)
    49→    49→  nextEpisode?: NextEpisodeInfo;
    50→    50→}
    51→    51→
    52→    52→type SubStatus = "idle" | "loading" | "loaded" | "none";
    53→    53→
    54→    54→// NOTE: no `sandbox` attribute on player iframes — VidSrc-family providers
    55→    55→// detect sandboxed frames and refuse to play ("Please Disable Sandbox").
    56→    56→// We keep the frame locked down via `allow` permissions instead.
    57→    57→
    58→    58→// ── Auto lead (Frame-Lock ±) ──
    59→    59→// Provider telemetry is sampled BEFORE it travels the relay chain
    60→    60→// (player → wrapper iframe → our window) and is quantized to 0.1s, so the
    61→    61→// reconstructed clock structurally trails the audio by ~300–500ms. The
    62→    62→// subtitle lookup compensates with a constant lead — the transport bar and
    63→    63→// watch-history stay exact. Per-episode [ / ] nudges stack on top.
    64→    64→const SUBTITLE_LEAD_S = 0.4;

    88→    88→  const [frameKey, setFrameKey] = useState(0);
    89→    89→  const [loading, setLoading] = useState(true);

    93→    93→
    94→    94→  // ── True fullscreen (stage-level) ──
    95→    95→  // The whole stage — provider video + subtitle overlay — goes fullscreen as
    96→    96→  // ONE element, so Arabic captions keep rendering above the video instead of
    97→    97→  // vanishing (the old bug: the provider iframe became the fullscreen element
    98→    98→  // and our sibling overlay disappeared). Provider native fullscreen is
    99→    99→  // stolen back for the stage; browsers without element fullscreen (iOS
   100→   100→  // Safari) get a fixed-position CSS stage instead.
   101→   101→  const stageRef = useRef<HTMLDivElement | null>(null);
   102→   102→  const [realFs, setRealFs] = useState(false);
   103→   103→  const [fakeFs, setFakeFs] = useState(false);
   104→   104→  const fsActive = realFs || fakeFs;
   105→   105→
   106→   106→  // ── per-episode sync memory ──
   107→   107→  // Every episode is a different rip with its own timing personality — a
   108→   108→  // +0.3s fix on S8E21 must not poison S8E22. The player remounts per
   109→   109→  // episode (keyed on imdb+s+e in the watch pages), so a lazy read is safe.
   110→   110→  // null → global default applies.
   111→   111→  const [epOffset, setEpOffset] = useState<number | null>(() =>
   112→   112→    readSubOffset({ type, imdbId, season, episode })
   113→   113→  );
   114→   114→  const effectiveOffset = epOffset ?? subOffset;
   115→   115→
   116→   116→  const applyOffset = useCallback(
   117→   117→    (v: number | null) => {
   118→   118→      setEpOffset(v);
   119→   119→      writeSubOffset({ type, imdbId, season, episode }, v);
   120→   120→    },
   121→   121→    [type, imdbId, season, episode]
   122→   122→  );
   123→   123→
   124→   124→  const nudgeOffset = useCallback(
   125→   125→    (d: number): number => {
   126→   126→      const next = Math.max(
   127→   127→        -30,
   128→   128→        Math.min(30, Math.round(((epOffset ?? subOffset) + d) * 10) / 10)
   129→   129→      );
   130→   130→      applyOffset(next);
   131→   131→      return next;
   132→   132→    },
   133→   133→    [epOffset, subOffset, applyOffset]
   134→   134→  );
   135→   135→
   136→   136→  // failover bookkeeping (refs — only touched inside effects/handlers)
   137→   137→  const loadedRef = useRef(false);
   138→   138→  const triedRef = useRef<Set<string>>(new Set());
   139→   139→  const userPickedRef = useRef(false);
   140→   140→
   141→   141→  // ── Autopilot telemetry engine ──
   142→   142→  const frameToken = `${serverId}-${frameKey}`;
   143→   143→  const playerFrameRef = useRef<HTMLIFrameElement | null>(null);
   144→   144→  const sync = usePlayerSync(frameToken, playerFrameRef);
   145→   145→
   146→   146→  // ── subtitle engine state ──
   147→   147→  const [cues, setCues] = useState<SubtitleCue[]>([]);
   148→   148→  const [subStatus, setSubStatus] = useState<SubStatus>("idle");
   149→   149→  const [subProgress, setSubProgress] = useState<number | null>(null);
   150→   150→  const [loadedLang, setLoadedLang] = useState<"ar" | "en" | null>(null);
   151→   151→  const [localCues, setLocalCues] = useState<SubtitleCue[] | null>(null);
   152→   152→  const localCuesRef = useRef<SubtitleCue[] | null>(null);
   153→   153→
   154→   154→  // ── autopilot: derived visibility (no mirrored state → no setState in effect) ──
   155→   155→  const [autopilotDismissed, setAutopilotDismissed] = useState(false);
   156→   156→  const autopilotFinished =
   157→   157→    !!nextEpisode &&
   158→   158→    (sync.status === "completed" ||
   159→   159→      (sync.duration > 120 &&
   160→   160→        sync.status === "playing" &&
   161→   161→        sync.currentTime >= sync.duration * 0.965));
   162→   162→  const showAutopilot = autopilotFinished && !autopilotDismissed;
   163→   163→
   164→   164→  const server = getServer(serverId);
   165→   165→  const src = server.build(type, imdbId, season, episode);
   166→   166→
   167→   167→  const fullTitle =
   168→   168→    type === "series" && season && episode
   169→   169→      ? `${title} · ${t("season")} ${season}, ${t("episode")} ${episode}`
   170→   170→      : title;
   171→   171→
   172→   172→  const resetPlaybackState = () => {
   173→   173→    setLoading(true);
   174→   174→    setSlow(false);
   175→   175→    setOverlayDismissed(false);
   176→   176→  };
   177→   177→
   178→   178→  // ── watchdog: 8s veil-cap → 20s auto-failover (per server, once) ──
   179→   179→  useEffect(() => {
   180→   180→    loadedRef.current = false;
   181→   181→    const veilT = setTimeout(() => {
   182→   182→      // never hold the video hostage — reveal whatever the provider rendered
   183→   183→      setLoading(false);
   184→   184→      if (!loadedRef.current) setSlow(true);
   185→   185→    }, 8000);
   186→   186→    const failT = setTimeout(() => {
   187→   187→      if (loadedRef.current || userPickedRef.current) return;
   188→   188→      triedRef.current.add(serverId);
   189→   189→      const next = SERVERS.find((s) => !triedRef.current.has(s.id));
   190→   190→      if (next) {
   191→   191→        toast({
   192→   192→          title: t("serverSlow"),
   193→   193→          description: `${server.name} ${server.tag ?? ""} → ${next.name} ${next.tag ?? ""}`,
   194→   194→        });
   195→   195→        setServerId(next.id);
   196→   196→        setPreferredServer(next.id);
   197→   197→        setFrameKey((k) => k + 1);
   198→   198→        resetPlaybackState();
   199→   199→      } else {
   200→   200→        setSlow(true); // every server tried — leave manual switcher
   201→   201→      }
   202→   202→    }, 20000);
   203→   203→    return () => {
   204→   204→      clearTimeout(veilT);
   205→   205→      clearTimeout(failT);
   206→   206→    };
   207→   207→  }, [serverId, frameKey]);
   208→   208→
   209→   209→  // ── subtitle fetching (Arabic default → English fallback preference) ──
   210→   210→  // Server may respond { pending: true } while the AI translation job runs;
   211→   211→  // poll every 5 s (up to ~4 min) until cues are ready.
   212→   212→  const [subFetchNonce, setSubFetchNonce] = useState(0);
   213→   213→  useEffect(() => {
   214→   214→    if (subLang === "off") return;
   215→   215→    let cancelled = false;
   216→   216→    let pollT: ReturnType<typeof setTimeout> | null = null;
   217→   217→    const loadingT = setTimeout(() => !cancelled && setSubStatus("loading"), 0);
   218→   218→    // toast only when real work happened (translation/polling or manual refetch);
   219→   219→    // instantly-cached tracks load silently — no notification noise per page view
   220→   220→    let sawWork = subFetchNonce > 0;
   221→   221→    (async () => {
   222→   222→      for (let attempt = 0; attempt < 48 && !cancelled; attempt++) {
   223→   223→        try {
   224→   224→          const qs = new URLSearchParams({ type, id: imdbId, lang: subLang === "ar" ? "ar" : "en" });
   225→   225→          if (type === "series" && season) qs.set("season", String(season));
   226→   226→          if (type === "series" && episode) qs.set("episode", String(episode));
   227→   227→          const res = await fetch(`/api/subtitles?${qs}`);
   228→   228→          const data = (await res.json()) as {
   229→   229→            cues?: SubtitleCue[];
   230→   230→            count?: number;
   231→   231→            pending?: boolean;
   232→   232→            progress?: number;
   233→   233→          };
   234→   234→          if (cancelled) return;
   235→   235→          if (data.pending) {
   236→   236→            sawWork = true;
   237→   237→            setSubProgress(typeof data.progress === "number" ? data.progress : null);
   238→   238→            // Arabic-only policy (user requirement): while the AI translation
   239→   239→            // cooks we show NO interim English track — silence until Arabic.
   240→   240→            setSubStatus("loading");
   241→   241→            await new Promise<void>((r) => {
   242→   242→              pollT = setTimeout(r, 5000);
   243→   243→            });
   244→   244→            continue;
   245→   245→          }
   246→   246→          const got = data.cues ?? [];
   247→   247→          if (got.length > 0) {
   248→   248→            if (!localCuesRef.current) setCues(got);
   249→   249→            setSubStatus("loaded");
   250→   250→            setSubProgress(null);
   251→   251→            setLoadedLang(subLang);
   252→   252→            if (sawWork) {
   253→   253→              toast({
   254→   254→                title: subLang === "ar" ? t("subsLoaded") : t("subsLoadedEn"),
   255→   255→                description: `${got.length.toLocaleString()} cues · ${t("appName")}`,
   256→   256→              });
   257→   257→            }
   258→   258→          } else {
   259→   259→            setSubStatus("none");
   260→   260→          }
   261→   261→          return;
   262→   262→        } catch {
   263→   263→          if (!cancelled) setSubStatus("none");
   264→   264→          return;
   265→   265→        }
   266→   266→      }
   267→   267→      if (!cancelled) setSubStatus("none");
   268→   268→    })();
   269→   269→    return () => {
   270→   270→      cancelled = true;
   271→   271→      clearTimeout(loadingT);
   272→   272→      if (pollT) clearTimeout(pollT);
   273→   273→    };
   274→   274→  }, [subLang, type, imdbId, season, episode, subFetchNonce]);
   275→   275→
   276→   276→  const refetchSubs = () => setSubFetchNonce((n) => n + 1);
   277→   277→
   278→   278→  const importFile = async (file: File) => {
   279→   279→    try {
   280→   280→      // charset-aware: CP1256 Arabic uploads decode correctly too
   281→   281→      const raw = decodeSubtitleBuffer(await file.arrayBuffer());
   282→   282→      const parsed = parseSubtitles(raw);
   283→   283→      if (parsed.length === 0) {
   284→   284→        setSubStatus("none");
   285→   285→        toast({ title: t("subsNone"), variant: "destructive" });
   286→   286→        return;
   287→   287→      }
   288→   288→      setLocalCues(parsed);
   289→   289→      localCuesRef.current = parsed;
   290→   290→      setCues(parsed);
   291→   291→      setSubStatus("loaded");
   292→   292→      setLoadedLang(parsed.some((c) => /[\u0600-\u06FF]/.test(c.text)) ? "ar" : "en");
   293→   293→      if (subLang === "off") setSubLang("ar");
   294→   294→      toast({ title: t("subsLoaded"), description: `${parsed.length.toLocaleString()} cues` });
   295→   295→    } catch {
   296→   296→      toast({ title: t("error"), variant: "destructive" });
   297→   297→    }
   298→   298→  };
   299→   299→
   300→   300→  // ── watch history: mark visit (keep prior progress), then exact telemetry ──
   301→   301→  const lastSaveRef = useRef(0);
   302→   302→  const lastPctRef = useRef(0);
   303→   303→
   304→   304→  const saveProgress = useCallback(
   305→   305→    (pct: number) => {
   306→   306→      lastPctRef.current = pct;
   307→   307→      lastSaveRef.current = Date.now();
   308→   308→      useHistory.getState().upsert({
   309→   309→        imdbId,
   310→   310→        type,
   311→   311→        title,
   312→   312→        poster,
   313→   313→        season,
   314→   314→        episode,
   315→   315→        episodeTitle,
   316→   316→        progressPct: Math.max(1, Math.min(100, Math.round(pct))),
   317→   317→      });
   318→   318→    },
   319→   319→    [imdbId, type, title, poster, season, episode, episodeTitle]
   320→   320→  );
   321→   321→
   322→   322→  useEffect(() => {
   323→   323→    // first visit marks 1%, re-visits keep their old progress
   324→   324→    const prior = useHistory.getState().get(imdbId);
   325→   325→    saveProgress(prior?.progressPct ?? 1);
   326→   326→  }, [imdbId, season, episode]);
   327→   327→
   328→   328→  // progress pipeline: exact from provider telemetry, estimate in manual mode
   329→   329→  useEffect(() => {
   330→   330→    if (sync.status === "completed") {
   331→   331→      saveProgress(100);
   332→   332→      return;
   333→   333→    }
   334→   334→    if (sync.providerDriven && sync.duration > 0 && sync.currentTime > 0) {
   335→   335→      const pct = (sync.currentTime / sync.duration) * 100;
   336→   336→      const due = Date.now() - lastSaveRef.current > 15_000;
   337→   337→      const milestone = Math.floor(pct / 5) > Math.floor(lastPctRef.current / 5);
   338→   338→      if (due || milestone) saveProgress(pct);
   339→   339→    } else if (sync.manualMode && sync.currentTime > 0) {
   340→   340→      if (Date.now() - lastSaveRef.current > 30_000) {
   341→   341→        const denom = (type === "series" ? 45 : runtimeMin ?? 110) * 60;
   342→   342→        saveProgress(Math.min(97, (sync.currentTime / denom) * 100));
   343→   343→      }
   344→   344→    }
   345→   345→  }, [sync.currentTime, sync.duration, sync.status, sync.providerDriven, sync.manualMode, type, runtimeMin]);
   346→   346→
   347→   347→  // persist 100% exactly once at completion

   350→   350→    if (autopilotFinished && !autopilotSavedRef.current) {
   351→   351→      autopilotSavedRef.current = true;
   352→   352→      saveProgress(100);
   353→   353→    }
   354→   354→    if (!autopilotFinished) autopilotSavedRef.current = false;

   356→   356→

   365→   365→    setFrameKey((k) => k + 1);

   370→   370→    setFrameKey((k) => k + 1);
   371→   371→    resetPlaybackState();
   372→   372→  };
   373→   373→
   374→   374→  // last-resort escape hatch: provider pages always play in a top-level tab

   376→   376→    window.open(src, "_blank", "noopener");
   377→   377→  };
   378→   378→

   381→   381→    toast({ title: t("syncRunning"), description: t("startSyncHint") });
   382→   382→  };
   383→   383→
   384→   384→  // ── True fullscreen control ──
   385→   385→  const enterFullscreen = useCallback(() => {
   386→   386→    const el = stageRef.current;
   387→   387→    if (!el) return;
   388→   388→    const webkitReq = (el as HTMLDivElement & { webkitRequestFullscreen?: () => Promise<void> })
   389→   389→      .webkitRequestFullscreen;
   390→   390→    const req = el.requestFullscreen?.bind(el) ?? webkitReq?.bind(el);
   391→   391→    if (req) {
   392→   392→      req().catch(() => setFakeFs(true));
   393→   393→    } else {
   394→   394→      setFakeFs(true); // iOS Safari: element fullscreen unsupported → CSS stage
   395→   395→    }
   396→   396→  }, []);
   397→   397→
   398→   398→  const exitFullscreen = useCallback(() => {
   399→   399→    setFakeFs(false);
   400→   400→    const webkitDoc = document as Document & { webkitExitFullscreen?: () => void };
   401→   401→    if (document.fullscreenElement) {
   402→   402→      (document.exitFullscreen ?? webkitDoc.webkitExitFullscreen)?.call(document);
   403→   403→    }
   404→   404→  }, []);
   405→   405→
   406→   406→  const toggleFullscreen = useCallback(() => {
   407→   407→    if (realFs || fakeFs) exitFullscreen();
   408→   408→    else enterFullscreen();
   409→   409→  }, [realFs, fakeFs, enterFullscreen, exitFullscreen]);
   410→   410→
   411→   411→  // fullscreen tracking + provider-steal: if the provider iframe grabs
   412→   412→  // fullscreen (its own ⛶ button), swap it for our stage so the subtitle
   413→   413→  // overlay — a sibling of the iframe — stays in the picture. Works while
   414→   414→  // the document is fullscreen without a fresh user gesture; hard-fallback
   415→   415→  // is a CSS stage.
   416→   416→  useEffect(() => {
   417→   417→    const onFsChange = () => {
   418→   418→      const fe = document.fullscreenElement;
   419→   419→      setRealFs(!!fe && fe === stageRef.current);
   420→   420→      if (fe && fe === playerFrameRef.current && stageRef.current) {
   421→   421→        const stage = stageRef.current;
   422→   422→        void (async () => {
   423→   423→          try {
   424→   424→            await stage.requestFullscreen();
   425→   425→          } catch {
   426→   426→            try {
   427→   427→              await document.exitFullscreen();
   428→   428→            } catch {
   429→   429→              /* already out */
   430→   430→            }
   431→   431→            try {
   432→   432→              await stage.requestFullscreen();
   433→   433→            } catch {
   434→   434→              /* gesture-policy denied — clean exit below */
   435→   435→            }
   436→   436→          }
   437→   437→          // if the steal didn't take (browser gesture policy), leave the
   438→   438→          // provider's fullscreen entirely — our ⛶ button re-enters at stage
   439→   439→          // level, where subtitles live
   440→   440→          setTimeout(() => {
   441→   441→            if (document.fullscreenElement === playerFrameRef.current) {
   442→   442→              document.exitFullscreen?.().catch(() => {});
   443→   443→            }
   444→   444→          }, 400);
   445→   445→        })();
   446→   446→      }
   447→   447→    };
   448→   448→    document.addEventListener("fullscreenchange", onFsChange);
   449→   449→    document.addEventListener("webkitfullscreenchange", onFsChange);
   450→   450→    return () => {
   451→   451→      document.removeEventListener("fullscreenchange", onFsChange);
   452→   452→      document.removeEventListener("webkitfullscreenchange", onFsChange);
   453→   453→    };
