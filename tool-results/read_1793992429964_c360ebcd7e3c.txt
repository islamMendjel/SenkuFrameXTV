    34→// ─── StreamPlayer · multi-server player w/ Autopilot precision sync ──────────
    35→//
    36→// VidSrc-family providers post PLAYER_EVENT telemetry (currentTime / duration
    37→// / play-state) up the relay chain to our window. usePlayerSync turns that
    38→// into a live clock → perfectly-synced subtitles, an exact transport bar and
    39→// the end-of-episode Autopilot card. Servers that don't report fall back to
    40→// a one-tap manual sync clock.
    41→
    42→interface Props {
    43→  type: MediaType;
    44→  imdbId: string;
    45→  title: string;
    46→  season?: number;
    47→  episode?: number;
    48→  episodeTitle?: string;
    49→  poster?: string;
    50→  runtimeMin?: number; // for progress estimation (manual mode)
    51→  nextEpisode?: NextEpisodeInfo;
    52→}
    53→
    54→type SubStatus = "idle" | "loading" | "loaded" | "none";
    55→
    56→// ── Ad shield (iframe sandbox) ──
    57→// VidSrc-family embeds monetize HARD: popup-unders, tab-unders, clickaway
    58→// redirects. A sandboxed frame WITHOUT allow-popups / allow-top-navigation
    59→// kills every one of them at UA level — spec-guaranteed, no extension needed.
    60→// Providers that break inside a sandbox are marked sandboxOk:false, and any
    61→// server that fails WHILE shielded is retried unsandboxed once — the shield
    62→// never costs us playback (self-healing exception, remembered per server).
    63→function sandboxException(id: string): boolean {
    64→  try {
    65→    return (
    66→      typeof window !== "undefined" &&
    67→      window.localStorage.getItem(`senku.nosandbox:${id}`) === "1"
    68→    );
    69→  } catch {
    70→    return false;
    71→  }
    72→}
    73→function markSandboxException(id: string) {
    74→  try {
    75→    window.localStorage.setItem(`senku.nosandbox:${id}`, "1");
    76→  } catch {
    77→    /* private mode — in-memory retry only */
    78→  }
    79→}

    81→// ── Auto lead (Frame-Lock ±) ──
    82→// The RELEASE offset (~1s, user-measured) is calibrated at the DATA layer —
    83→// /api/subtitles + /api/subs/file serve cues shifted by SUB_CALIBRATION_S, so
    84→// both the provider-native track and this overlay inherit it. What remains
    85→// HERE is only the relay-chain latency: telemetry is sampled before it
    86→// travels the player → wrapper → our window, so the reconstructed clock
    87→// trails the audio by ~250ms. Per-episode [ / ] nudges and TapSync stack on
    88→// top of both.
    89→const SUBTITLE_LEAD_S = 0.25;

   109→    setPreferredServer,
   110→    setSubLang,

   113→  const [frameKey, setFrameKey] = useState(0);

   129→  // vanishing (the old bug: the provider iframe became the fullscreen element

   133→  const stageRef = useRef<HTMLDivElement | null>(null);

   159→        -30,

   165→    [epOffset, subOffset, applyOffset]

   170→  const triedRef = useRef<Set<string>>(new Set());
   171→  const userPickedRef = useRef(false);
   172→
   173→  // ── Autopilot telemetry engine ──
   174→  const frameToken = `${serverId}-${frameKey}-${sandboxOff ? "ns" : "sb"}`;
   175→  const playerFrameRef = useRef<HTMLIFrameElement | null>(null);
   176→  const sync = usePlayerSync(frameToken, playerFrameRef);
   177→
   178→  // ── subtitle engine state ──
   179→  const [cues, setCues] = useState<SubtitleCue[]>([]);
   180→  const [subStatus, setSubStatus] = useState<SubStatus>("idle");
   181→  const [subProgress, setSubProgress] = useState<number | null>(null);
   182→  const [loadedLang, setLoadedLang] = useState<"ar" | "en" | null>(null);
   183→  const [localCues, setLocalCues] = useState<SubtitleCue[] | null>(null);
   184→  const localCuesRef = useRef<SubtitleCue[] | null>(null);
   185→
   186→  // ── autopilot: derived visibility (no mirrored state → no setState in effect) ──
   187→  const [autopilotDismissed, setAutopilotDismissed] = useState(false);
   188→  const autopilotFinished =
   189→    !!nextEpisode &&
   190→    (sync.status === "completed" ||
   191→      (sync.duration > 120 &&
   192→        sync.status === "playing" &&
   193→        sync.currentTime >= sync.duration * 0.965));
   194→  const showAutopilot = autopilotFinished && !autopilotDismissed;
   195→
   196→  const server = getServer(serverId);
   197→
   198→  // ── Provider-native Arabic track (VidSrc ds_lang/sub_url injection) ──
   199→  // When a cached Arabic track exists, it is handed to the provider as an
   200→  // auto-applied external subtitle (sub_url). The provider then renders OUR
   201→  // cues on its own video timeline — relay-chain latency disappears, the
   202→  // English default track never surfaces, and captions live inside the video
   203→  // so they survive every fullscreen mode. The 60fps overlay stays as the
   204→  // fallback path (servers without injection / no cached track yet).
   205→  const supportsNativeSubs = server.nativeSubs === true;
   206→  const [nativeSubReady, setNativeSubReady] = useState<boolean | null>(
   207→    supportsNativeSubs ? null : false
   208→  );
   209→  const nativeProbeEpoch = useRef(0);
   210→  useEffect(() => {
   211→    const epoch = ++nativeProbeEpoch.current;
   212→    void (async () => {
   213→      await Promise.resolve(); // microtask — effect body stays setState-free
   214→      if (nativeProbeEpoch.current !== epoch) return;
   215→      if (!supportsNativeSubs) {
   216→        setNativeSubReady(false);
   217→        return;
   218→      }
   219→      setNativeSubReady(null);
   220→      try {
   221→        const qs = new URLSearchParams({ type, id: imdbId, lang: "ar" });
   222→        if (type === "series" && season) qs.set("season", String(season));
   223→        if (type === "series" && episode) qs.set("episode", String(episode));
   224→        const res = await fetch(`/api/subs/available?${qs}`);
   225→        const data = (await res.json()) as { exists?: boolean };
   226→        if (nativeProbeEpoch.current !== epoch) return;
   227→        setNativeSubReady(!!data.exists);
   228→      } catch {
   229→        if (nativeProbeEpoch.current !== epoch) return;
   230→        setNativeSubReady(false);
   231→      }
   232→    })();
   233→    return () => {
   234→      nativeProbeEpoch.current += 1;
   235→    };
   236→  }, [supportsNativeSubs, type, imdbId, season, episode]);
   237→
   238→  // absolute https URL required — the PROVIDER iframe fetches it from the
   239→  // viewer's browser, so a relative path would resolve against their origin
   240→  const subFileUrl = nativeSubReady
   241→    ? `${window.location.origin}/api/subs/file?${new URLSearchParams({
   242→        type,
   243→        id: imdbId,
   244→        lang: "ar",
   245→        ...(type === "series" && season ? { season: String(season) } : {}),
   246→        ...(type === "series" && episode ? { episode: String(episode) } : {}),
   247→      }).toString()}`
   248→    : null;
   249→  // src stays "" (about:blank behind the loading veil) until the availability
   250→  // probe lands — one navigation, no reload flicker.
   251→  const src =
   252→    nativeSubReady === null
   253→      ? ""
   254→      : server.build(type, imdbId, season, episode, {
   255→          subUrl: subFileUrl ?? undefined,
   256→        });
   257→
   258→  // ── AR-only policy (user requirement) ──
   259→  // Playback ALWAYS starts with Arabic subtitles on: an English/OFF selection
   260→  // left over from an earlier session is overridden the moment the player
   261→  // mounts (the player remounts per episode, so every episode gets AR).
   262→  useEffect(() => {
   263→    const s = useSettings.getState();
   264→    if (s.subLang !== "ar") s.setSubLang("ar");
   265→  }, []);
   266→
   267→  const fullTitle =
   268→    type === "series" && season && episode
   269→      ? `${title} · ${t("season")} ${season}, ${t("episode")} ${episode}`
   270→      : title;
   271→
   272→  const resetPlaybackState = () => {
   273→    setLoading(true);
   274→    setSlow(false);
   275→    setOverlayDismissed(false);
   276→  };
   277→
   278→  // ── watchdog: 8s veil-cap → 20s auto-failover (per server, once) ──
   279→  useEffect(() => {
   280→    loadedRef.current = false;
   281→    const veilT = setTimeout(() => {
   282→      // never hold the video hostage — reveal whatever the provider rendered
   283→      setLoading(false);
   284→      if (!loadedRef.current) setSlow(true);
   285→    }, 8000);
   286→    const failT = setTimeout(() => {
   287→      if (loadedRef.current || userPickedRef.current) return;
   288→      // self-heal: a server that can't even load inside the ad shield gets
   289→      // ONE unsandboxed retry before we give up on it entirely
   290→      if (server.sandboxOk !== false && !sandboxOff && !sandboxRetriedRef.current) {
   291→        sandboxRetriedRef.current = true;
   292→        markSandboxException(serverId);
   293→        setSandboxOff(true);
   294→        setFrameKey((k) => k + 1);
   295→        resetPlaybackState();
   296→        toast({ title: t("adsShieldRetry"), description: server.name });
   297→        return;
   298→      }
   299→      triedRef.current.add(serverId);
   300→      const next = SERVERS.find((s) => !triedRef.current.has(s.id));
   301→      if (next) {
   302→        toast({
   303→          title: t("serverSlow"),
   304→          description: `${server.name} ${server.tag ?? ""} → ${next.name} ${next.tag ?? ""}`,
   305→        });
   306→        setServerId(next.id);
   307→        setPreferredServer(next.id);
   308→        setSandboxOff(sandboxException(next.id));
   309→        sandboxRetriedRef.current = false;
   310→        setFrameKey((k) => k + 1);
   311→        resetPlaybackState();
   312→      } else {
   313→        setSlow(true); // every server tried — leave manual switcher
   314→      }
   315→    }, 20000);
   316→    return () => {
   317→      clearTimeout(veilT);
   318→      clearTimeout(failT);
   319→    };
   320→  }, [serverId, frameKey, server, sandboxOff]);
   321→
   322→  // ── subtitle fetching (Arabic default → English fallback preference) ──
   323→  // Server may respond { pending: true } while the AI translation job runs;
   324→  // poll every 5 s (up to ~4 min) until cues are ready.
   325→  const [subFetchNonce, setSubFetchNonce] = useState(0);
   326→  useEffect(() => {
   327→    if (subLang === "off") return;
   328→    let cancelled = false;
   329→    let pollT: ReturnType<typeof setTimeout> | null = null;
   330→    const loadingT = setTimeout(() => !cancelled && setSubStatus("loading"), 0);
   331→    // toast only when real work happened (translation/polling or manual refetch);
   332→    // instantly-cached tracks load silently — no notification noise per page view
   333→    let sawWork = subFetchNonce > 0;
   334→    (async () => {
   335→      for (let attempt = 0; attempt < 48 && !cancelled; attempt++) {
   336→        try {
   337→          const qs = new URLSearchParams({ type, id: imdbId, lang: subLang === "ar" ? "ar" : "en" });
   338→          if (type === "series" && season) qs.set("season", String(season));
   339→          if (type === "series" && episode) qs.set("episode", String(episode));
   340→          const res = await fetch(`/api/subtitles?${qs}`);
   341→          const data = (await res.json()) as {
   342→            cues?: SubtitleCue[];
   343→            count?: number;
   344→            pending?: boolean;
   345→            progress?: number;
   346→          };
   347→          if (cancelled) return;
   348→          if (data.pending) {
   349→            sawWork = true;
   350→            setSubProgress(typeof data.progress === "number" ? data.progress : null);
   351→            // Arabic-only policy (user requirement): while the AI translation
   352→            // cooks we show NO interim English track — silence until Arabic.
   353→            setSubStatus("loading");
   354→            await new Promise<void>((r) => {
   355→              pollT = setTimeout(r, 5000);
   356→            });
   357→            continue;
   358→          }
   359→          const got = data.cues ?? [];
   360→          if (got.length > 0) {
   361→            if (!localCuesRef.current) setCues(got);
   362→            setSubStatus("loaded");
   363→            setSubProgress(null);
   364→            setLoadedLang(subLang);
   365→            if (sawWork) {
   366→              toast({
   367→                title: subLang === "ar" ? t("subsLoaded") : t("subsLoadedEn"),
   368→                description: `${got.length.toLocaleString()} cues · ${t("appName")}`,
   369→              });
   370→            }
   371→          } else {
   372→            setSubStatus("none");
   373→          }
   374→          return;
   375→        } catch {
   376→          if (!cancelled) setSubStatus("none");
   377→          return;
   378→        }
   379→      }
   380→      if (!cancelled) setSubStatus("none");
   381→    })();
   382→    return () => {
   383→      cancelled = true;
   384→      clearTimeout(loadingT);
   385→      if (pollT) clearTimeout(pollT);
   386→    };
   387→  }, [subLang, type, imdbId, season, episode, subFetchNonce]);
   388→
   389→  const refetchSubs = () => setSubFetchNonce((n) => n + 1);
   390→
   391→  const importFile = async (file: File) => {
   392→    try {
   393→      // charset-aware: CP1256 Arabic uploads decode correctly too
   394→      const raw = decodeSubtitleBuffer(await file.arrayBuffer());
   395→      const parsed = parseSubtitles(raw);
   396→      if (parsed.length === 0) {
   397→        setSubStatus("none");
   398→        toast({ title: t("subsNone"), variant: "destructive" });
   399→        return;
   400→      }
   401→      setLocalCues(parsed);
   402→      localCuesRef.current = parsed;
   403→      setCues(parsed);
   404→      setSubStatus("loaded");
   405→      setLoadedLang(parsed.some((c) => /[\u0600-\u06FF]/.test(c.text)) ? "ar" : "en");
   406→      if (subLang === "off") setSubLang("ar");
   407→      toast({ title: t("subsLoaded"), description: `${parsed.length.toLocaleString()} cues` });
   408→    } catch {
   409→      toast({ title: t("error"), variant: "destructive" });
   410→    }
   411→  };
   412→
   413→  // ── watch history: mark visit (keep prior progress), then exact telemetry ──
   414→  const lastSaveRef = useRef(0);
   415→  const lastPctRef = useRef(0);
   416→
   417→  const saveProgress = useCallback(
   418→    (pct: number) => {
   419→      lastPctRef.current = pct;
   420→      lastSaveRef.current = Date.now();
   421→      useHistory.getState().upsert({
   422→        imdbId,
   423→        type,
   424→        title,
   425→        poster,
   426→        season,
   427→        episode,
   428→        episodeTitle,
   429→        progressPct: Math.max(1, Math.min(100, Math.round(pct))),
   430→      });
   431→    },
   432→    [imdbId, type, title, poster, season, episode, episodeTitle]
   433→  );
   434→
   435→  useEffect(() => {
   436→    // first visit marks 1%, re-visits keep their old progress
   437→    const prior = useHistory.getState().get(imdbId);
   438→    saveProgress(prior?.progressPct ?? 1);
   439→  }, [imdbId, season, episode]);
   440→
   441→  // progress pipeline: exact from provider telemetry, estimate in manual mode
   442→  useEffect(() => {
   443→    if (sync.status === "completed") {
   444→      saveProgress(100);
   445→      return;
   446→    }
   447→    if (sync.providerDriven && sync.duration > 0 && sync.currentTime > 0) {
   448→      const pct = (sync.currentTime / sync.duration) * 100;
   449→      const due = Date.now() - lastSaveRef.current > 15_000;
   450→      const milestone = Math.floor(pct / 5) > Math.floor(lastPctRef.current / 5);
   451→      if (due || milestone) saveProgress(pct);
   452→    } else if (sync.manualMode && sync.currentTime > 0) {
   453→      if (Date.now() - lastSaveRef.current > 30_000) {
   454→        const denom = (type === "series" ? 45 : runtimeMin ?? 110) * 60;
   455→        saveProgress(Math.min(97, (sync.currentTime / denom) * 100));
   456→      }
   457→    }
   458→  }, [sync.currentTime, sync.duration, sync.status, sync.providerDriven, sync.manualMode, type, runtimeMin]);
   459→
   460→  // persist 100% exactly once at completion
   461→  const autopilotSavedRef = useRef(false);
   462→  useEffect(() => {
   463→    if (autopilotFinished && !autopilotSavedRef.current) {
   464→      autopilotSavedRef.current = true;
   465→      saveProgress(100);
   466→    }
   467→    if (!autopilotFinished) autopilotSavedRef.current = false;
   468→  }, [autopilotFinished, saveProgress]);
   469→
   470→  // ── Autopilot trigger bookkeeping is derived (see autopilotFinished above) ──
   471→
   472→  const switchServer = (id: string) => {
   473→    if (id === serverId) return;
   474→    userPickedRef.current = true; // manual choice → disable auto-failover
   475→    triedRef.current = new Set([id]);
   476→    setServerId(id);
   477→    setPreferredServer(id);
   478→    setSandboxOff(sandboxException(id));
   479→    sandboxRetriedRef.current = false;
   480→    setFrameKey((k) => k + 1);
   481→    resetPlaybackState();
   482→  };
   483→
   484→  const refreshFrame = () => {
   485→    setFrameKey((k) => k + 1);
   486→    resetPlaybackState();
   487→  };
   488→
   489→  // last-resort escape hatch: provider pages always play in a top-level tab
   490→  const openDirect = () => {
   491→    window.open(src, "_blank", "noopener");
   492→  };
   493→
   494→  const startManualSync = () => {
   495→    sync.armManual();
   496→    toast({ title: t("syncRunning"), description: t("startSyncHint") });
   497→  };
   498→
   499→  // ── TapSync™ — one-tap ear calibration ──
   500→  // The viewer taps exactly when they HEAR a line; the caption that should be
   501→  // showing (the active one if it's already up, else the next upcoming cue)
   502→  // snaps to that instant. One tap = exact correction, self-replaceable —
   503→  // no modal, no ±0.1 hunting. Persisted per episode like every nudge.
   504→  const tapSync = useCallback(() => {
   505→    const list = localCues ?? cues;
   506→    if (list.length === 0) {
   507→      toast({ title: t("tapSyncNone"), variant: "destructive" });
   508→      return;
   509→    }
   510→    const tTap = sync.getTime();
   511→    const lookup = tTap + effectiveOffset + SUBTITLE_LEAD_S;
   512→    const target = cueAt(list, lookup) ?? list.find((c) => c.start > lookup) ?? null;
   513→    if (!target) {
   514→      toast({ title: t("tapSyncNone"), variant: "destructive" });
   515→      return;
   516→    }
   517→    // want: target shows exactly at tap → tTap + epOffset + LEAD = target.start
   518→    const next = Math.max(
   519→      -30,
   520→      Math.min(30, Math.round((target.start - tTap - SUBTITLE_LEAD_S) * 10) / 10)
   521→    );
   522→    applyOffset(next);
   523→    toast({
   524→      title: t("tapSynced"),
