    32→    32→// ─── StreamPlayer · multi-server player w/ Autopilot precision sync ──────────
    33→    33→//
    34→    34→// VidSrc-family providers post PLAYER_EVENT telemetry (currentTime / duration
    35→    35→// / play-state) up the relay chain to our window. usePlayerSync turns that
    36→    36→// into a live clock → perfectly-synced subtitles, an exact transport bar and
    37→    37→// the end-of-episode Autopilot card. Servers that don't report fall back to
    38→    38→// a one-tap manual sync clock.
    39→    39→
    40→    40→interface Props {
    41→    41→  type: MediaType;
    42→    42→  imdbId: string;
    43→    43→  title: string;
    44→    44→  season?: number;
    45→    45→  episode?: number;
    46→    46→  episodeTitle?: string;
    47→    47→  poster?: string;
    48→    48→  runtimeMin?: number; // for progress estimation (manual mode)
    49→    49→  nextEpisode?: NextEpisodeInfo;
    50→    50→}
    51→    51→
    52→    52→type SubStatus = "idle" | "loading" | "loaded" | "none";
    53→    53→
    54→    54→// NOTE: no `sandbox` attribute on player iframes — VidSrc-family providers
    55→    55→// detect sandboxed frames and refuse to play ("Please Disable Sandbox").
    56→    56→// We keep the frame locked down via `allow` permissions instead.
    57→    57→
    58→    58→// ── Auto lead (Frame-Lock ±) ──
    59→    59→// Provider telemetry is sampled BEFORE it travels the relay chain
    60→    60→// (player → wrapper iframe → our window) and is quantized to 0.1s, so the
    61→    61→// reconstructed clock structurally trails the audio by ~300–500ms. The
    62→    62→// subtitle lookup compensates with a constant lead — the transport bar and
    63→    63→// watch-history stay exact. Per-episode [ / ] nudges stack on top.
    64→    64→const SUBTITLE_LEAD_S = 0.4;

    94→    94→  // ── True fullscreen (stage-level) ──

   168→   168→  // auto-applied external subtitle (sub_url). The provider then renders OUR

   172→   172→  // fallback path (servers without injection / no cached track yet).
   173→   173→  const supportsNativeSubs = server.nativeSubs === true;
   174→   174→  const [nativeSubReady, setNativeSubReady] = useState<boolean | null>(

   181→   181→      await Promise.resolve(); // microtask — effect body stays setState-free
   182→   182→      if (nativeProbeEpoch.current !== epoch) return;
   183→   183→      if (!supportsNativeSubs) {
   184→   184→        setNativeSubReady(false);
   185→   185→        return;
   186→   186→      }
   187→   187→      setNativeSubReady(null);

   202→   202→      nativeProbeEpoch.current += 1;

   210→   210→        type,
   211→   211→        id: imdbId,

   213→   213→        ...(type === "series" && season ? { season: String(season) } : {}),
   214→   214→        ...(type === "series" && episode ? { episode: String(episode) } : {}),

   220→   220→    nativeSubReady === null
   221→   221→      ? ""
   222→   222→      : server.build(type, imdbId, season, episode, {
   223→   223→          subUrl: subFileUrl ?? undefined,

   225→   225→

   227→   227→  // Playback ALWAYS starts with Arabic subtitles on: an English/OFF selection

   310→   310→              pollT = setTimeout(r, 5000);

   322→   322→                title: subLang === "ar" ? t("subsLoaded") : t("subsLoadedEn"),
   323→   323→                description: `${got.length.toLocaleString()} cues · ${t("appName")}`,
   324→   324→              });
   325→   325→            }
   326→   326→          } else {
   327→   327→            setSubStatus("none");
   328→   328→          }
   329→   329→          return;
   330→   330→        } catch {
   331→   331→          if (!cancelled) setSubStatus("none");
   332→   332→          return;
   333→   333→        }
   334→   334→      }
   335→   335→      if (!cancelled) setSubStatus("none");
   336→   336→    })();
   337→   337→    return () => {
   338→   338→      cancelled = true;
   339→   339→      clearTimeout(loadingT);
   340→   340→      if (pollT) clearTimeout(pollT);
   341→   341→    };
   342→   342→  }, [subLang, type, imdbId, season, episode, subFetchNonce]);
   343→   343→

   345→   345→

   366→   366→  };
   367→   367→
   368→   368→  // ── watch history: mark visit (keep prior progress), then exact telemetry ──
   369→   369→  const lastSaveRef = useRef(0);

   371→   371→

   377→   377→        imdbId,
   378→   378→        type,
   379→   379→        title,
   380→   380→        poster,
   381→   381→        season,
   382→   382→        episode,
   383→   383→        episodeTitle,
   384→   384→        progressPct: Math.max(1, Math.min(100, Math.round(pct))),
   385→   385→      });
   386→   386→    },
   387→   387→    [imdbId, type, title, poster, season, episode, episodeTitle]
   388→   388→  );
   389→   389→
   390→   390→  useEffect(() => {
   391→   391→    // first visit marks 1%, re-visits keep their old progress

   393→   393→    saveProgress(prior?.progressPct ?? 1);
   394→   394→  }, [imdbId, season, episode]);
   395→   395→
   396→   396→  // progress pipeline: exact from provider telemetry, estimate in manual mode
   397→   397→  useEffect(() => {
   398→   398→    if (sync.status === "completed") {
   399→   399→      saveProgress(100);
   400→   400→      return;
   401→   401→    }
   402→   402→    if (sync.providerDriven && sync.duration > 0 && sync.currentTime > 0) {
   403→   403→      const pct = (sync.currentTime / sync.duration) * 100;
   404→   404→      const due = Date.now() - lastSaveRef.current > 15_000;
   405→   405→      const milestone = Math.floor(pct / 5) > Math.floor(lastPctRef.current / 5);
   406→   406→      if (due || milestone) saveProgress(pct);
   407→   407→    } else if (sync.manualMode && sync.currentTime > 0) {
   408→   408→      if (Date.now() - lastSaveRef.current > 30_000) {
   409→   409→        const denom = (type === "series" ? 45 : runtimeMin ?? 110) * 60;
   410→   410→        saveProgress(Math.min(97, (sync.currentTime / denom) * 100));
   411→   411→      }
   412→   412→    }
   413→   413→  }, [sync.currentTime, sync.duration, sync.status, sync.providerDriven, sync.manualMode, type, runtimeMin]);
   414→   414→
   415→   415→  // persist 100% exactly once at completion
   416→   416→  const autopilotSavedRef = useRef(false);
   417→   417→  useEffect(() => {
   418→   418→    if (autopilotFinished && !autopilotSavedRef.current) {
   419→   419→      autopilotSavedRef.current = true;
   420→   420→      saveProgress(100);
   421→   421→    }
   422→   422→    if (!autopilotFinished) autopilotSavedRef.current = false;
   423→   423→  }, [autopilotFinished, saveProgress]);
   424→   424→
   425→   425→  // ── Autopilot trigger bookkeeping is derived (see autopilotFinished above) ──
   426→   426→
   427→   427→  const switchServer = (id: string) => {

   429→   429→    userPickedRef.current = true; // manual choice → disable auto-failover
   430→   430→    triedRef.current = new Set([id]);
   431→   431→    setServerId(id);
   432→   432→    setPreferredServer(id);
   433→   433→    setFrameKey((k) => k + 1);
   434→   434→    resetPlaybackState();
   435→   435→  };
   436→   436→
   437→   437→  const refreshFrame = () => {
   438→   438→    setFrameKey((k) => k + 1);
   439→   439→    resetPlaybackState();
   440→   440→  };
   441→   441→
   442→   442→  // last-resort escape hatch: provider pages always play in a top-level tab
   443→   443→  const openDirect = () => {
   444→   444→    window.open(src, "_blank", "noopener");
