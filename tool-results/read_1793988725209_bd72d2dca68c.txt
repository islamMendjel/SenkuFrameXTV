    32→// ─── StreamPlayer · multi-server player w/ Autopilot precision sync ──────────
    33→//
    34→// VidSrc-family providers post PLAYER_EVENT telemetry (currentTime / duration
    35→// / play-state) up the relay chain to our window. usePlayerSync turns that
    36→// into a live clock → perfectly-synced subtitles, an exact transport bar and
    37→// the end-of-episode Autopilot card. Servers that don't report fall back to
    38→// a one-tap manual sync clock.
    39→
    40→interface Props {
    41→  type: MediaType;
    42→  imdbId: string;
    43→  title: string;
    44→  season?: number;
    45→  episode?: number;
    46→  episodeTitle?: string;
    47→  poster?: string;
    48→  runtimeMin?: number; // for progress estimation (manual mode)
    49→  nextEpisode?: NextEpisodeInfo;
    50→}
    51→
    52→type SubStatus = "idle" | "loading" | "loaded" | "none";
    53→
    54→// NOTE: no `sandbox` attribute on player iframes — VidSrc-family providers
    55→// detect sandboxed frames and refuse to play ("Please Disable Sandbox").
    56→// We keep the frame locked down via `allow` permissions instead.
    57→
    58→// ── Auto lead (Frame-Lock ±) ──
    59→// Provider telemetry is sampled BEFORE it travels the relay chain
    60→// (player → wrapper iframe → our window) and is quantized to 0.1s, so the
    61→// reconstructed clock structurally trails the audio by ~300–500ms. The
    62→// subtitle lookup compensates with a constant lead — the transport bar and
    63→// watch-history stay exact. Per-episode [ / ] nudges stack on top.
    64→const SUBTITLE_LEAD_S = 0.4;

   173→  const supportsNativeSubs = server.nativeSubs === true;

   220→    nativeSubReady === null

   222→      : server.build(type, imdbId, season, episode, {
   223→          subUrl: subFileUrl ?? undefined,
   224→        });
   225→
   226→  // ── AR-only policy (user requirement) ──
   227→  // Playback ALWAYS starts with Arabic subtitles on: an English/OFF selection

   229→  // mounts (the player remounts per episode, so every episode gets AR).
   230→  useEffect(() => {
   231→    const s = useSettings.getState();

   233→  }, []);
   234→
   235→  const fullTitle =
   236→    type === "series" && season && episode
   237→      ? `${title} · ${t("season")} ${season}, ${t("episode")} ${episode}`
   238→      : title;
   239→
   240→  const resetPlaybackState = () => {
   241→    setLoading(true);
   242→    setSlow(false);
   243→    setOverlayDismissed(false);
   244→  };
   245→
   246→  // ── watchdog: 8s veil-cap → 20s auto-failover (per server, once) ──
   247→  useEffect(() => {
   248→    loadedRef.current = false;
   249→    const veilT = setTimeout(() => {
   250→      // never hold the video hostage — reveal whatever the provider rendered
   251→      setLoading(false);
   252→      if (!loadedRef.current) setSlow(true);
   253→    }, 8000);
   254→    const failT = setTimeout(() => {
   255→      if (loadedRef.current || userPickedRef.current) return;
   256→      triedRef.current.add(serverId);
   257→      const next = SERVERS.find((s) => !triedRef.current.has(s.id));
   258→      if (next) {
   259→        toast({
   260→          title: t("serverSlow"),
   261→          description: `${server.name} ${server.tag ?? ""} → ${next.name} ${next.tag ?? ""}`,
   262→        });

   264→        setPreferredServer(next.id);
   265→        setFrameKey((k) => k + 1);
   266→        resetPlaybackState();
   267→      } else {
   268→        setSlow(true); // every server tried — leave manual switcher
   269→      }
   270→    }, 20000);
   271→    return () => {
   272→      clearTimeout(veilT);
   273→      clearTimeout(failT);
   274→    };
   275→  }, [serverId, frameKey]);

   277→  // ── subtitle fetching (Arabic default → English fallback preference) ──
   278→  // Server may respond { pending: true } while the AI translation job runs;
   279→  // poll every 5 s (up to ~4 min) until cues are ready.
   280→  const [subFetchNonce, setSubFetchNonce] = useState(0);
   281→  useEffect(() => {
   282→    if (subLang === "off") return;
   283→    let cancelled = false;
   284→    let pollT: ReturnType<typeof setTimeout> | null = null;
   285→    const loadingT = setTimeout(() => !cancelled && setSubStatus("loading"), 0);
   286→    // toast only when real work happened (translation/polling or manual refetch);
   287→    // instantly-cached tracks load silently — no notification noise per page view
   288→    let sawWork = subFetchNonce > 0;
   289→    (async () => {
   290→      for (let attempt = 0; attempt < 48 && !cancelled; attempt++) {
   291→        try {
   292→          const qs = new URLSearchParams({ type, id: imdbId, lang: subLang === "ar" ? "ar" : "en" });
   293→          if (type === "series" && season) qs.set("season", String(season));
   294→          if (type === "series" && episode) qs.set("episode", String(episode));
   295→          const res = await fetch(`/api/subtitles?${qs}`);
   296→          const data = (await res.json()) as {
   297→            cues?: SubtitleCue[];
   298→            count?: number;
   299→            pending?: boolean;
   300→            progress?: number;
   301→          };
   302→          if (cancelled) return;
   303→          if (data.pending) {
   304→            sawWork = true;
   305→            setSubProgress(typeof data.progress === "number" ? data.progress : null);
   306→            // Arabic-only policy (user requirement): while the AI translation
   307→            // cooks we show NO interim English track — silence until Arabic.
   308→            setSubStatus("loading");
   309→            await new Promise<void>((r) => {
   310→              pollT = setTimeout(r, 5000);
   311→            });
   312→            continue;
   313→          }
   314→          const got = data.cues ?? [];
   315→          if (got.length > 0) {
   316→            if (!localCuesRef.current) setCues(got);
   317→            setSubStatus("loaded");
   318→            setSubProgress(null);
   319→            setLoadedLang(subLang);
   320→            if (sawWork) {
   321→              toast({
   322→                title: subLang === "ar" ? t("subsLoaded") : t("subsLoadedEn"),
   323→                description: `${got.length.toLocaleString()} cues · ${t("appName")}`,
   324→              });
   325→            }
   326→          } else {
   327→            setSubStatus("none");
   328→          }
   329→          return;
   330→        } catch {
   331→          if (!cancelled) setSubStatus("none");
   332→          return;
   333→        }
   334→      }
   335→      if (!cancelled) setSubStatus("none");
   336→    })();
   337→    return () => {
   338→      cancelled = true;
   339→      clearTimeout(loadingT);
   340→      if (pollT) clearTimeout(pollT);
   341→    };
   342→  }, [subLang, type, imdbId, season, episode, subFetchNonce]);
   343→
   344→  const refetchSubs = () => setSubFetchNonce((n) => n + 1);
   345→
   346→  const importFile = async (file: File) => {
   347→    try {
   348→      // charset-aware: CP1256 Arabic uploads decode correctly too

   351→      if (parsed.length === 0) {

   357→      localCuesRef.current = parsed;

   362→      toast({ title: t("subsLoaded"), description: `${parsed.length.toLocaleString()} cues` });
   363→    } catch {
   364→      toast({ title: t("error"), variant: "destructive" });
   365→    }
   366→  };
   367→
   368→  // ── watch history: mark visit (keep prior progress), then exact telemetry ──
   369→  const lastSaveRef = useRef(0);
   370→  const lastPctRef = useRef(0);
   371→
   372→  const saveProgress = useCallback(
   373→    (pct: number) => {
   374→      lastPctRef.current = pct;
   375→      lastSaveRef.current = Date.now();
   376→      useHistory.getState().upsert({
   377→        imdbId,
   378→        type,
   379→        title,
   380→        poster,
   381→        season,
   382→        episode,
   383→        episodeTitle,
   384→        progressPct: Math.max(1, Math.min(100, Math.round(pct))),
   385→      });
   386→    },
   387→    [imdbId, type, title, poster, season, episode, episodeTitle]
   388→  );
   389→
   390→  useEffect(() => {
   391→    // first visit marks 1%, re-visits keep their old progress
   392→    const prior = useHistory.getState().get(imdbId);
   393→    saveProgress(prior?.progressPct ?? 1);
   394→  }, [imdbId, season, episode]);
   395→

   398→    if (sync.status === "completed") {
   399→      saveProgress(100);
   400→      return;
   401→    }

   403→      const pct = (sync.currentTime / sync.duration) * 100;
   404→      const due = Date.now() - lastSaveRef.current > 15_000;
   405→      const milestone = Math.floor(pct / 5) > Math.floor(lastPctRef.current / 5);
   406→      if (due || milestone) saveProgress(pct);

   411→      }
   412→    }
   413→  }, [sync.currentTime, sync.duration, sync.status, sync.providerDriven, sync.manualMode, type, runtimeMin]);

   445→  };

   448→    sync.armManual();

   450→  };
   451→
   452→  // ── True fullscreen control ──
   453→  // requestFullscreen runs SYNCHRONOUSLY inside the click (transient user
   454→  // activation) and falls back through: webkit API → one delayed retry →
   455→  // CSS stage (fakeFs). The CSS stage guarantees a fullscreen-like view even
   456→  // when the UA denies the API outright (permission policies, sandboxed
   457→  // embedders, iOS Safari) — the user ALWAYS lands fullscreen.
   458→  const enterFullscreen = useCallback(() => {
   459→    const stage = stageRef.current;
   460→    if (!stage) return;
   461→    if (document.fullscreenElement === stage) return;
   462→    const stageAny = stage as HTMLDivElement & {
   463→      webkitRequestFullscreen?: () => Promise<void> | void;
   464→    };
   465→    const hasApi =
   466→      typeof stage.requestFullscreen === "function" ||
   467→      typeof stageAny.webkitRequestFullscreen === "function";
   468→    if (!hasApi) {
   469→      setFakeFs(true); // element fullscreen unsupported → CSS stage
   470→      return;
   471→    }
   472→    const req = (): Promise<void> | void =>
   473→      typeof stage.requestFullscreen === "function"
   474→        ? stage.requestFullscreen({ navigationUI: "hide" })
   475→        : stageAny.webkitRequestFullscreen?.();
   476→    try {
   477→      const p = req();
   478→      if (p && typeof (p as Promise<void>).catch === "function") {
   479→        (p as Promise<void>).catch(() => {
   480→          // denied — one beat later, one retry; then the CSS stage so the
   481→          // promise of a fullscreen view is always kept
   482→          setTimeout(() => {
   483→            if (document.fullscreenElement === stage) return;
   484→            try {
   485→              const p2 = req();
   486→              if (p2 && typeof (p2 as Promise<void>).catch === "function") {
   487→                (p2 as Promise<void>).catch(() => {
   488→                  setFakeFs(true);
   489→                  toast({ title: t("fullscreen"), description: t("fullscreenFallback") });
   490→                });
   491→              }
   492→            } catch {
   493→              setFakeFs(true);
   494→            }
   495→          }, 150);
   496→        });
   497→      }
   498→    } catch {
   499→      setFakeFs(true);
   500→    }
   501→  }, [t]);
   502→
   503→  const exitFullscreen = useCallback(() => {
   504→    setFakeFs(false);
   505→    const webkitDoc = document as Document & { webkitExitFullscreen?: () => void };
   506→    if (document.fullscreenElement) {
   507→      (document.exitFullscreen ?? webkitDoc.webkitExitFullscreen)?.call(document);
   508→    }
   509→  }, []);
   510→
   511→  const toggleFullscreen = useCallback(() => {
   512→    if (realFs || fakeFs) exitFullscreen();
   513→    else enterFullscreen();
   514→  }, [realFs, fakeFs, enterFullscreen, exitFullscreen]);
   515→
   516→  // fullscreen tracking + provider-steal: if the provider iframe grabs
   517→  // fullscreen (its own ⛶ button), swap it for our stage so the subtitle
   518→  // overlay — a sibling of the iframe — stays in the picture. While the
   519→  // document is already fullscreen, switching the fullscreen element to our
   520→  // stage needs NO fresh user gesture (nested switch); retries absorb
   521→  // transient denials, and a final exit leaves our page in front. A request
   522→  // rejected at UA level (fullscreenerror) flips to the CSS stage.
   523→  useEffect(() => {
   524→    const onFsChange = () => {
   525→      const doc = document as Document & { webkitFullscreenElement?: Element | null };
   526→      const fe = document.fullscreenElement ?? doc.webkitFullscreenElement ?? null;
   527→      setRealFs(!!fe && fe === stageRef.current);
   528→      if (fe && fe !== stageRef.current && stageRef.current) {
   529→        const stage = stageRef.current;
   530→        let tries = 0;
   531→        const steal = () => {
   532→          tries += 1;
