    32→// ─── StreamPlayer · multi-server player w/ Autopilot precision sync ──────────
    33→//
    34→// VidSrc-family providers post PLAYER_EVENT telemetry (currentTime / duration
    35→// / play-state) up the relay chain to our window. usePlayerSync turns that
    36→// into a live clock → perfectly-synced subtitles, an exact transport bar and
    37→// the end-of-episode Autopilot card. Servers that don't report fall back to
    38→// a one-tap manual sync clock.
    39→
    40→interface Props {
    41→  type: MediaType;
    42→  imdbId: string;
    43→  title: string;
    44→  season?: number;
    45→  episode?: number;
    46→  episodeTitle?: string;
    47→  poster?: string;
    48→  runtimeMin?: number; // for progress estimation (manual mode)
    49→  nextEpisode?: NextEpisodeInfo;
    50→}
    51→
    52→type SubStatus = "idle" | "loading" | "loaded" | "none";
    53→
    54→// NOTE: no `sandbox` attribute on player iframes — VidSrc-family providers
    55→// detect sandboxed frames and refuse to play ("Please Disable Sandbox").
    56→// We keep the frame locked down via `allow` permissions instead.
    57→
    58→// ── Auto lead (Frame-Lock ±) ──
    59→// Provider telemetry is sampled BEFORE it travels the relay chain
    60→// (player → wrapper iframe → our window) and is quantized to 0.1s, so the
    61→// reconstructed clock structurally trails the audio by ~300–500ms. The
    62→// subtitle lookup compensates with a constant lead — the transport bar and
    63→// watch-history stay exact. Per-episode [ / ] nudges stack on top.
    64→const SUBTITLE_LEAD_S = 0.4;
    65→
    66→export function StreamPlayer({
    67→  type,
    68→  imdbId,
    69→  title,
    70→  season,
    71→  episode,
    72→  episodeTitle,
    73→  poster,
    74→  runtimeMin,
    75→  nextEpisode,
    76→}: Props) {
    77→  const t = useT();
    78→  const {
    79→    subLang,
    80→    subOffset,
    81→    subSize,
    82→    subStyle,
    83→    preferredServer,
    84→    setPreferredServer,
    85→    setSubLang,
    86→  } = useSettings();
    87→  const [serverId, setServerId] = useState<string>(() => preferredServer ?? SERVERS[0].id);
    88→  const [frameKey, setFrameKey] = useState(0);
    89→  const [loading, setLoading] = useState(true);
    90→  const [slow, setSlow] = useState(false);
    91→  const [settingsOpen, setSettingsOpen] = useState(false);
    92→  const [overlayDismissed, setOverlayDismissed] = useState(false);
    93→
    94→  // ── True fullscreen (stage-level) ──
    95→  // The whole stage — provider video + subtitle overlay — goes fullscreen as
    96→  // ONE element, so Arabic captions keep rendering above the video instead of
    97→  // vanishing (the old bug: the provider iframe became the fullscreen element
    98→  // and our sibling overlay disappeared). Provider native fullscreen is
    99→  // stolen back for the stage; browsers without element fullscreen (iOS
   100→  // Safari) get a fixed-position CSS stage instead.
   101→  const stageRef = useRef<HTMLDivElement | null>(null);
   102→  const [realFs, setRealFs] = useState(false);
   103→  const [fakeFs, setFakeFs] = useState(false);
   104→  const fsActive = realFs || fakeFs;
   105→
   106→  // ── per-episode sync memory ──
   107→  // Every episode is a different rip with its own timing personality — a
   108→  // +0.3s fix on S8E21 must not poison S8E22. The player remounts per
   109→  // episode (keyed on imdb+s+e in the watch pages), so a lazy read is safe.
   110→  // null → global default applies.
   111→  const [epOffset, setEpOffset] = useState<number | null>(() =>
   112→    readSubOffset({ type, imdbId, season, episode })
   113→  );
   114→  const effectiveOffset = epOffset ?? subOffset;
   115→
   116→  const applyOffset = useCallback(
   117→    (v: number | null) => {
   118→      setEpOffset(v);
   119→      writeSubOffset({ type, imdbId, season, episode }, v);
   120→    },
   121→    [type, imdbId, season, episode]
   122→  );
   123→
   124→  const nudgeOffset = useCallback(
   125→    (d: number): number => {
   126→      const next = Math.max(
   127→        -30,
   128→        Math.min(30, Math.round(((epOffset ?? subOffset) + d) * 10) / 10)
   129→      );
   130→      applyOffset(next);
   131→      return next;
   132→    },
   133→    [epOffset, subOffset, applyOffset]
   134→  );
   135→
   136→  // failover bookkeeping (refs — only touched inside effects/handlers)
   137→  const loadedRef = useRef(false);
   138→  const triedRef = useRef<Set<string>>(new Set());
   139→  const userPickedRef = useRef(false);
   140→
   141→  // ── Autopilot telemetry engine ──
   142→  const frameToken = `${serverId}-${frameKey}`;
   143→  const playerFrameRef = useRef<HTMLIFrameElement | null>(null);
   144→  const sync = usePlayerSync(frameToken, playerFrameRef);
   145→
   146→  // ── subtitle engine state ──
   147→  const [cues, setCues] = useState<SubtitleCue[]>([]);
   148→  const [subStatus, setSubStatus] = useState<SubStatus>("idle");
   149→  const [subProgress, setSubProgress] = useState<number | null>(null);
   150→  const [loadedLang, setLoadedLang] = useState<"ar" | "en" | null>(null);
   151→  const [localCues, setLocalCues] = useState<SubtitleCue[] | null>(null);
   152→  const localCuesRef = useRef<SubtitleCue[] | null>(null);
   153→
   154→  // ── autopilot: derived visibility (no mirrored state → no setState in effect) ──
   155→  const [autopilotDismissed, setAutopilotDismissed] = useState(false);
   156→  const autopilotFinished =
   157→    !!nextEpisode &&
   158→    (sync.status === "completed" ||
   159→      (sync.duration > 120 &&
   160→        sync.status === "playing" &&
   161→        sync.currentTime >= sync.duration * 0.965));
   162→  const showAutopilot = autopilotFinished && !autopilotDismissed;
   163→
   164→  const server = getServer(serverId);
   165→  const src = server.build(type, imdbId, season, episode);
   166→
   167→  const fullTitle =
   168→    type === "series" && season && episode
   169→      ? `${title} · ${t("season")} ${season}, ${t("episode")} ${episode}`
   170→      : title;
   171→
   172→  const resetPlaybackState = () => {
   173→    setLoading(true);
   174→    setSlow(false);
   175→    setOverlayDismissed(false);
   176→  };
   177→
   178→  // ── watchdog: 8s veil-cap → 20s auto-failover (per server, once) ──
   179→  useEffect(() => {
   180→    loadedRef.current = false;
   181→    const veilT = setTimeout(() => {
   182→      // never hold the video hostage — reveal whatever the provider rendered
   183→      setLoading(false);
   184→      if (!loadedRef.current) setSlow(true);
   185→    }, 8000);
   186→    const failT = setTimeout(() => {
   187→      if (loadedRef.current || userPickedRef.current) return;
   188→      triedRef.current.add(serverId);
   189→      const next = SERVERS.find((s) => !triedRef.current.has(s.id));
   190→      if (next) {
   191→        toast({
   192→          title: t("serverSlow"),
   193→          description: `${server.name} ${server.tag ?? ""} → ${next.name} ${next.tag ?? ""}`,
   194→        });
   195→        setServerId(next.id);
   196→        setPreferredServer(next.id);
   197→        setFrameKey((k) => k + 1);
   198→        resetPlaybackState();
   199→      } else {
   200→        setSlow(true); // every server tried — leave manual switcher
   201→      }
   202→    }, 20000);
   203→    return () => {
   204→      clearTimeout(veilT);
   205→      clearTimeout(failT);
   206→    };
   207→  }, [serverId, frameKey]);
   208→
   209→  // ── subtitle fetching (Arabic default → English fallback preference) ──
   210→  // Server may respond { pending: true } while the AI translation job runs;
   211→  // poll every 5 s (up to ~4 min) until cues are ready.
   212→  const [subFetchNonce, setSubFetchNonce] = useState(0);
   213→  useEffect(() => {
   214→    if (subLang === "off") return;
   215→    let cancelled = false;
   216→    let pollT: ReturnType<typeof setTimeout> | null = null;
   217→    const loadingT = setTimeout(() => !cancelled && setSubStatus("loading"), 0);
   218→    // toast only when real work happened (translation/polling or manual refetch);
   219→    // instantly-cached tracks load silently — no notification noise per page view
   220→    let sawWork = subFetchNonce > 0;
   221→    (async () => {
   222→      for (let attempt = 0; attempt < 48 && !cancelled; attempt++) {
   223→        try {
   224→          const qs = new URLSearchParams({ type, id: imdbId, lang: subLang === "ar" ? "ar" : "en" });
   225→          if (type === "series" && season) qs.set("season", String(season));
   226→          if (type === "series" && episode) qs.set("episode", String(episode));
   227→          const res = await fetch(`/api/subtitles?${qs}`);
   228→          const data = (await res.json()) as {
   229→            cues?: SubtitleCue[];
   230→            count?: number;
   231→            pending?: boolean;
   232→            progress?: number;
   233→          };
   234→          if (cancelled) return;
   235→          if (data.pending) {
   236→            sawWork = true;
   237→            setSubProgress(typeof data.progress === "number" ? data.progress : null);
   238→            // Arabic-only policy (user requirement): while the AI translation
   239→            // cooks we show NO interim English track — silence until Arabic.
   240→            setSubStatus("loading");
   241→            await new Promise<void>((r) => {
   242→              pollT = setTimeout(r, 5000);
   243→            });
   244→            continue;
   245→          }
   246→          const got = data.cues ?? [];
   247→          if (got.length > 0) {
   248→            if (!localCuesRef.current) setCues(got);
   249→            setSubStatus("loaded");
   250→            setSubProgress(null);
   251→            setLoadedLang(subLang);
   252→            if (sawWork) {
   253→              toast({
   254→                title: subLang === "ar" ? t("subsLoaded") : t("subsLoadedEn"),
   255→                description: `${got.length.toLocaleString()} cues · ${t("appName")}`,
   256→              });
   257→            }
   258→          } else {
   259→            setSubStatus("none");
   260→          }
   261→          return;
   262→        } catch {
   263→          if (!cancelled) setSubStatus("none");
   264→          return;
   265→        }
   266→      }
   267→      if (!cancelled) setSubStatus("none");
   268→    })();
   269→    return () => {
   270→      cancelled = true;
   271→      clearTimeout(loadingT);
   272→      if (pollT) clearTimeout(pollT);
   273→    };
   274→  }, [subLang, type, imdbId, season, episode, subFetchNonce]);
   275→
   276→  const refetchSubs = () => setSubFetchNonce((n) => n + 1);
   277→
   278→  const importFile = async (file: File) => {
   279→    try {
   280→      // charset-aware: CP1256 Arabic uploads decode correctly too
   281→      const raw = decodeSubtitleBuffer(await file.arrayBuffer());
   282→      const parsed = parseSubtitles(raw);
   283→      if (parsed.length === 0) {
   284→        setSubStatus("none");
   285→        toast({ title: t("subsNone"), variant: "destructive" });
   286→        return;
   287→      }
   288→      setLocalCues(parsed);
   289→      localCuesRef.current = parsed;
   290→      setCues(parsed);
   291→      setSubStatus("loaded");
   292→      setLoadedLang(parsed.some((c) => /[\u0600-\u06FF]/.test(c.text)) ? "ar" : "en");
   293→      if (subLang === "off") setSubLang("ar");
   294→      toast({ title: t("subsLoaded"), description: `${parsed.length.toLocaleString()} cues` });
   295→    } catch {
   296→      toast({ title: t("error"), variant: "destructive" });
   297→    }
   298→  };
   299→
   300→  // ── watch history: mark visit (keep prior progress), then exact telemetry ──
   301→  const lastSaveRef = useRef(0);
   302→  const lastPctRef = useRef(0);
   303→
   304→  const saveProgress = useCallback(
   305→    (pct: number) => {
   306→      lastPctRef.current = pct;
   307→      lastSaveRef.current = Date.now();
   308→      useHistory.getState().upsert({
   309→        imdbId,
   310→        type,
   311→        title,
   312→        poster,
   313→        season,
   314→        episode,
   315→        episodeTitle,
   316→        progressPct: Math.max(1, Math.min(100, Math.round(pct))),
   317→      });
   318→    },
   319→    [imdbId, type, title, poster, season, episode, episodeTitle]
   320→  );
   321→
   322→  useEffect(() => {
   323→    // first visit marks 1%, re-visits keep their old progress
   324→    const prior = useHistory.getState().get(imdbId);
   325→    saveProgress(prior?.progressPct ?? 1);
   326→  }, [imdbId, season, episode]);
   327→
   328→  // progress pipeline: exact from provider telemetry, estimate in manual mode
   329→  useEffect(() => {
   330→    if (sync.status === "completed") {
   331→      saveProgress(100);
   332→      return;
   333→    }
   334→    if (sync.providerDriven && sync.duration > 0 && sync.currentTime > 0) {
   335→      const pct = (sync.currentTime / sync.duration) * 100;
   336→      const due = Date.now() - lastSaveRef.current > 15_000;
   337→      const milestone = Math.floor(pct / 5) > Math.floor(lastPctRef.current / 5);
   338→      if (due || milestone) saveProgress(pct);
   339→    } else if (sync.manualMode && sync.currentTime > 0) {
   340→      if (Date.now() - lastSaveRef.current > 30_000) {
   341→        const denom = (type === "series" ? 45 : runtimeMin ?? 110) * 60;
   342→        saveProgress(Math.min(97, (sync.currentTime / denom) * 100));
   343→      }
   344→    }
   345→  }, [sync.currentTime, sync.duration, sync.status, sync.providerDriven, sync.manualMode, type, runtimeMin]);
   346→
   347→  // persist 100% exactly once at completion
   348→  const autopilotSavedRef = useRef(false);
   349→  useEffect(() => {
   350→    if (autopilotFinished && !autopilotSavedRef.current) {
   351→      autopilotSavedRef.current = true;
   352→      saveProgress(100);
   353→    }
   354→    if (!autopilotFinished) autopilotSavedRef.current = false;
   355→  }, [autopilotFinished, saveProgress]);
   356→
   357→  // ── Autopilot trigger bookkeeping is derived (see autopilotFinished above) ──
   358→
   359→  const switchServer = (id: string) => {
   360→    if (id === serverId) return;
   361→    userPickedRef.current = true; // manual choice → disable auto-failover
   362→    triedRef.current = new Set([id]);
   363→    setServerId(id);
   364→    setPreferredServer(id);
   365→    setFrameKey((k) => k + 1);
   366→    resetPlaybackState();
   367→  };
   368→
   369→  const refreshFrame = () => {
   370→    setFrameKey((k) => k + 1);
   371→    resetPlaybackState();
   372→  };
   373→
   374→  // last-resort escape hatch: provider pages always play in a top-level tab
   375→  const openDirect = () => {
   376→    window.open(src, "_blank", "noopener");
   377→  };
   378→
   379→  const startManualSync = () => {
   380→    sync.armManual();
   381→    toast({ title: t("syncRunning"), description: t("startSyncHint") });
   382→  };
   383→
   384→  // ── True fullscreen control ──
   385→  const enterFullscreen = useCallback(() => {
   386→    const el = stageRef.current;
   387→    if (!el) return;
   388→    const webkitReq = (el as HTMLDivElement & { webkitRequestFullscreen?: () => Promise<void> })
   389→      .webkitRequestFullscreen;
   390→    const req = el.requestFullscreen?.bind(el) ?? webkitReq?.bind(el);
   391→    if (req) {
   392→      req().catch(() => setFakeFs(true));
   393→    } else {
   394→      setFakeFs(true); // iOS Safari: element fullscreen unsupported → CSS stage
   395→    }
   396→  }, []);
   397→
   398→  const exitFullscreen = useCallback(() => {
   399→    setFakeFs(false);
   400→    const webkitDoc = document as Document & { webkitExitFullscreen?: () => void };
   401→    if (document.fullscreenElement) {
   402→      (document.exitFullscreen ?? webkitDoc.webkitExitFullscreen)?.call(document);
   403→    }
   404→  }, []);
   405→
   406→  const toggleFullscreen = useCallback(() => {
   407→    if (realFs || fakeFs) exitFullscreen();
   408→    else enterFullscreen();
   409→  }, [realFs, fakeFs, enterFullscreen, exitFullscreen]);
   410→
   411→  // fullscreen tracking + provider-steal: if the provider iframe grabs
   412→  // fullscreen (its own ⛶ button), swap it for our stage so the subtitle
   413→  // overlay — a sibling of the iframe — stays in the picture. Works while
   414→  // the document is fullscreen without a fresh user gesture; hard-fallback
   415→  // is a CSS stage.
   416→  useEffect(() => {
   417→    const onFsChange = () => {
   418→      const fe = document.fullscreenElement;
   419→      setRealFs(!!fe && fe === stageRef.current);
   420→      if (fe && fe === playerFrameRef.current && stageRef.current) {
   421→        const stage = stageRef.current;
   422→        void (async () => {
   423→          try {
   424→            await stage.requestFullscreen();
   425→          } catch {
   426→            try {
   427→              await document.exitFullscreen();
   428→            } catch {
   429→              /* already out */
   430→            }
   431→            try {
   432→              await stage.requestFullscreen();
   433→            } catch {
   434→              /* gesture-policy denied — clean exit below */
   435→            }
   436→          }
   437→          // if the steal didn't take (browser gesture policy), leave the
   438→          // provider's fullscreen entirely — our ⛶ button re-enters at stage
   439→          // level, where subtitles live
   440→          setTimeout(() => {
   441→            if (document.fullscreenElement === playerFrameRef.current) {
   442→              document.exitFullscreen?.().catch(() => {});
   443→            }
   444→          }, 400);
   445→        })();
   446→      }
   447→    };
   448→    document.addEventListener("fullscreenchange", onFsChange);
   449→    document.addEventListener("webkitfullscreenchange", onFsChange);
   450→    return () => {
   451→      document.removeEventListener("fullscreenchange", onFsChange);
   452→      document.removeEventListener("webkitfullscreenchange", onFsChange);
   453→    };
   454→  }, []);
   455→
   456→  // keyboard: F = fullscreen · [ = subs later · ] = subs earlier
   457→  useEffect(() => {
   458→    const onKey = (e: KeyboardEvent) => {
   459→      const el = e.target as HTMLElement | null;
   460→      if (
   461→        el &&
   462→        (/^(input|textarea|select)$/i.test(el.tagName) || el.isContentEditable)
   463→      )
   464→        return;
   465→      if (e.metaKey || e.ctrlKey || e.altKey) return;
   466→      if (e.key === "f" || e.key === "F") {
   467→        e.preventDefault();
   468→        toggleFullscreen();
   469→      } else if (e.key === "[") {
   470→        const v = nudgeOffset(-0.1);
   471→        toast({
   472→          title: t("sync"),
   473→          description: `${formatOffset(v * 1000)} · ${t("offsetLater")}`,
   474→        });
   475→      } else if (e.key === "]") {
   476→        const v = nudgeOffset(0.1);
   477→        toast({
   478→          title: t("sync"),
   479→          description: `${formatOffset(v * 1000)} · ${t("offsetEarlier")}`,
   480→        });
   481→      }
   482→    };
   483→    window.addEventListener("keydown", onKey);
   484→    return () => window.removeEventListener("keydown", onKey);
   485→  });
   486→
   487→  const cueCount = (localCues ?? cues).length;
   488→  const showSubs = subLang !== "off" && subStatus === "loaded";
   489→  const translatingAr = subLang === "ar" && subStatus === "loading";
   490→  const progressPct =
   491→    sync.duration > 0 ? Math.min(100, (sync.currentTime / sync.duration) * 100) : 0;
   492→
   493→  // sync-mode chip: precision (live) / estimated / manual
   494→  const syncChip = !showSubs ? null : sync.providerDriven ? (
   495→    <span
   496→      className={cn(
   497→        "flex min-h-[36px] items-center gap-1.5 rounded-full border px-3 py-1.5 text-xs font-bold",
   498→        sync.estimated
   499→          ? "border-amber-400/50 bg-amber-400/10 text-amber-300"
   500→          : "border-primary/50 bg-primary/10 text-primary"
   501→      )}
   502→      title={sync.estimated ? t("estimatedSync") : t("precisionSync")}
   503→    >
   504→      <Radio className={cn("h-3.5 w-3.5", !sync.estimated && "animate-pulse-glow")} aria-hidden />
   505→      <span className="hidden sm:inline">
   506→        {sync.estimated ? t("estimatedSync") : t("precisionSync")}
   507→      </span>
   508→    </span>
   509→  ) : sync.manualMode ? (
   510→    <span className="flex min-h-[36px] items-center gap-1.5 rounded-full border border-amber-400/50 bg-amber-400/10 px-3 py-1.5 text-xs font-bold text-amber-300">
   511→      <Timer className="h-3.5 w-3.5" aria-hidden />
   512→      <span className="hidden sm:inline">{t("manualSync")}</span>
   513→    </span>
   514→  ) : null;
   515→
   516→  return (
   517→    <section className="w-full" aria-label={`${t("playerTitle")} ${fullTitle}`}>
   518→      {/* ── Video frame (fullscreen stage) ── */}
   519→      <div
   520→        ref={stageRef}
   521→        className={cn(
   522→          "relative aspect-video w-full overflow-hidden border-y border-border/60 bg-black sm:rounded-2xl sm:border",
   523→          fakeFs && "fixed inset-0 z-[90] aspect-auto h-[100dvh] w-screen"
   524→        )}
   525→        // inline style beats the sm: variants (Tailwind cascade) so the
   526→        // fullscreen stage is edge-to-edge: no rounded corners, no border
   527→        style={fsActive ? { borderRadius: 0, borderWidth: 0 } : undefined}
   528→      >
   529→        <iframe
   530→          key={frameToken}
   531→          src={src}
   532→          title={fullTitle}
   533→          ref={playerFrameRef}
   534→          className="absolute inset-0 h-full w-full"
   535→          referrerPolicy="origin"
